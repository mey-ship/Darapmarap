country_event = { #Poland Ruler Failsafe
	id = my_custom_event.3
	title = my_custom_event.3.t
	desc = my_custom_event.3.d
	picture = CIVIL_WAR_eventPicture

	fire_only_once = yes

	trigger = {
		tag = POL
		NOT = { has_reform = polish_elective_monarchy }
		NOT = { has_ruler = "Filip Agnieszka" }
		is_year = 1448
		NOT = { is_year = 1475 }
		is_free_or_tributary_trigger = yes
	}

	 immediate = {
		hidden_effect = {
			random_owned_province = {
				limit = {
					is_state = yes
					is_capital = no
					is_in_capital_area = yes
				}
				save_event_target_as = gecko_pol_pretenders
			}
		}
		hidden_effect = {
			random_owned_province = {
				limit = {
					is_state = yes
					is_capital = no
				}
				save_event_target_as = gecko_pol_nobles
			}
		}
		hidden_effect = {
			random_owned_province = {
				limit = {
					is_state = yes
					is_capital = no
				}
				save_event_target_as = gecko_pol_nobles2
			}
		}
	 }

	mean_time_to_happen = {
		months = 12
	}

	option = {
		name = my_custom_event.3.a
		add_estate_loyalty = {
			estate = estate_nobles
			loyalty = -20
		}
		define_ruler = {
			culture = ROOT
			religion = ROOT
			adm = 5
			dip = 4
			mil = 5
			age = 25
			claim = 20
		}
		event_target:gecko_pol_nobles = {
			spawn_rebels = {
				type = noble_rebels
				size = 2
			}
		}
	}

	option = {
		name = my_custom_event.3.b
		add_legitimacy = 20
	}
}
