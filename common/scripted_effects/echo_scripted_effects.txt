development_counter = {
	[[on_startup]
		if = {
			limit = {
				ai = no	
				NOT = { has_country_flag = has_development_counter_cflag }				
			}
			set_variable = {
				which = adm_dev_counter
				value = 0
			}
			set_variable = {
				which = dip_dev_counter
				value = 0
			}
			set_variable = {
				which = mil_dev_counter
				value = 0
			}
			set_variable = {
				which = total_dev_counter
				value = 0
			}
			set_country_flag = has_development_counter_cflag
		}
	]
	
	#All updates to variables in this section are in province scope.
	[[adm]
		owner = {
			if = {
				limit = { ai = no }
				change_variable = {
					which = adm_dev_counter
					value = 1
				}
				change_variable = {
					which = total_dev_counter
					value = 1
				}
			}
		}
	]
	[[dip]
		owner = {
			if = {
				limit = { ai = no }
				change_variable = {
					which = dip_dev_counter
					value = 1
				}
				change_variable = {
					which = total_dev_counter
					value = 1
				}
			}
		}
	]
	[[mil]
		owner = {
			if = {
				limit = { ai = no }
				change_variable = {
					which = mil_dev_counter
					value = 1
				}
				change_variable = {
					which = total_dev_counter
					value = 1
				}
			}
		}
	]
}
bastard_dev_cost_add_from_buildings = {
	if = {
		limit = {
			num_of_buildings_in_province = 1
			NOT = {num_of_buildings_in_province = 2}
		}
		add_province_modifier = {
				name = bastard_dev_cost_modifier_1
				duration = -1
		}
	}
	if = {
		limit = {
			num_of_buildings_in_province = 2
			NOT = {num_of_buildings_in_province = 3}
		}
		remove_province_modifier = bastard_dev_cost_modifier_1

				name = bastard_dev_cost_modifier_3

gc_update_expanded_infrastructure_bonus = {

	remove_province_modifier = gc_expanded_infrastructure_extra_1
	remove_province_modifier = gc_expanded_infrastructure_extra_2
	remove_province_modifier = gc_expanded_infrastructure_extra_3
	remove_province_modifier = gc_expanded_infrastructure_extra_4
	remove_province_modifier = gc_expanded_infrastructure_extra_5

	if = {
		limit = {
			owner = {
				full_idea_group = infrastructure_ideas
			}
		}

		if = {
			limit = {
				num_of_times_expanded_infrastructure = 5
			}
			add_province_modifier = {
				name = gc_expanded_infrastructure_extra_5
				duration = -1
			}
		}
		else_if = {
			limit = {
				num_of_times_expanded_infrastructure = 4
			}
			add_province_modifier = {
				name = gc_expanded_infrastructure_extra_4
				duration = -1
			}
		}
		else_if = {
			limit = {
				num_of_times_expanded_infrastructure = 3
			}
			add_province_modifier = {
				name = gc_expanded_infrastructure_extra_3
				duration = -1
			}
		}
		else_if = {
			limit = {
				num_of_times_expanded_infrastructure = 2
			}
			add_province_modifier = {
				name = gc_expanded_infrastructure_extra_2
				duration = -1
			}
		}
		else_if = {
			limit = {
				num_of_times_expanded_infrastructure = 1
			}
			add_province_modifier = {
				name = gc_expanded_infrastructure_extra_1
				duration = -1
			}
		}
	}
}
